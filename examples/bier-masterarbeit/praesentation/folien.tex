\section{Motivation}

\begin{frame}{Why Beer?}
  \begin{tugitemize}
    \item Only four ingredients -- yet thousands of aromas
    \item Much of the aroma is created in the \alert{fermenter}
    \item Systematic data on fermentation temperature are missing
  \end{tugitemize}
  \vspace{4mm}
  \begin{block}{Research Question}
    How strongly does fermentation temperature shape the aroma of top-fermented beers?
  \end{block}
\end{frame}

\section{Method}

\begin{frame}{Approach}
  \begin{tugcolumns}
    \begin{column}{.48\linewidth}
      \begin{tugenumerate}
        \item 12 brews of pale ale
        \item 4 temperature levels (16--24\,°C)
        \item Gas chromatography
        \item Blind tasting ($n=24$)
      \end{tugenumerate}
    \end{column}
    \begin{column}{.48\linewidth}
      \begin{block}{Arrhenius}
        \[ k(T) = A \cdot e^{-E_a / RT} \]
      \end{block}
    \end{column}
  \end{tugcolumns}
\end{frame}
